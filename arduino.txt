#include <ESP8266WiFi.h>
#include <PubSubClient.h>
#include <DHT.h>
#include <Servo.h>
#include <ArduinoJson.h>

// ==================== PIN DEFINITIONS ====================
#define DHTPIN D2
#define DHTTYPE DHT11

// Gas Storage Tank Section
#define ULTRASONIC_TRIG D6
#define ULTRASONIC_ECHO D7
#define MQ2_PIN A0              // Gas sensor (analog)

// Ventilation & Exhaust Section  
#define RELAY_FAN D1            // Relay for exhaust fan
#define SERVO_VENT D4           // Servo for vent flap

// Entry Security Section
#define PIR_ENTRY D5            // PIR at entry point
#define PIR_EXIT D8             // Optional second PIR

// Additional Sensors (for enhanced demo)
#define LIGHT_SENSOR A1         // Optional LDR
#define BUZZER_PIN D3           // Local alarm buzzer

// Device Configuration
const char* DEVICE_ID = "factory_floor_1";
const char* DEVICE_LOCATION = "Factory Floor A - Tank Section";
const char* DEVICE_TYPE = "industrial";

// WiFi & MQTT
const char* ssid = "BEYONDER 5430";
const char* password = "tejas143";
const char* mqtt_server = "10.136.75.54";   // Fog node IP
const int mqtt_port = 1883;

WiFiClient espClient;
PubSubClient client(espClient);

DHT dht(DHTPIN, DHTTYPE);
Servo ventServo;

// Sensor state tracking
unsigned long lastPublishTime = 0;
unsigned long heartbeatInterval = 5000;    // 5 seconds
unsigned long publishInterval = 2000;      // 2 seconds

// Alert state
bool gasAlertActive = false;
bool overflowAlertActive = false;
bool securityAlertActive = false;
unsigned long alertCooldown = 0;

// Calibration values
const float GAS_NORMAL_MAX = 300;
const float GAS_WARNING_MAX = 500;
const float GAS_CRITICAL_MAX = 700;
const float TANK_SAFE_DISTANCE = 10.0;     // cm from top
const float TANK_WARNING_DISTANCE = 5.0;   // cm from top

void setup_wifi() {
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500); 
    Serial.println(".");
  }

  Serial.println("Wifi Connected 🤝");
  Serial.println("IP Address: ");
  Serial.println(WiFi.localIP());
}

// ==================== MQTT CALLBACK ====================
void callback(char* topic, byte* payload, unsigned int length) {
  String msg;
  for (int i = 0; i < length; i++) msg += (char)payload[i];
  
  Serial.println("MQTT Message on " + String(topic) + ": " + msg);

  // Fan control
  if (String(topic) == "factory/actuator/fan") {
    if (msg == "ON") {
      digitalWrite(RELAY_FAN, LOW);  // Active LOW relay
      Serial.println("✅ FAN ON - Exhaust started");
    } else {
      digitalWrite(RELAY_FAN, HIGH);
      Serial.println("⏹ FAN OFF");
    }
  }

  // Vent control
  if (String(topic) == "factory/actuator/vent") {
    if (msg == "OPEN") {
      ventServo.write(90);
      Serial.println("✅ VENT OPEN - Airflow increased");
    } else {
      ventServo.write(0);
      Serial.println("⏹ VENT CLOSED");
    }
  }
  
  // Emergency override
  if (String(topic) == "fognetx/actuator/emergency") {
    if (msg == "RESET") {
      // Reset all alerts
      gas_alertActive = false;
      overflowAlertActive = false;
      securityAlertActive = false;
      digitalWrite(RELAY_FAN, HIGH);
      ventServo.write(0);
      digitalWrite(BUZZER_PIN, LOW);
      Serial.println("🔄 EMERGENCY RESET - All systems normal");
    }
  }
}

void reconnect() {
  while (!client.connected()) {
    client.connect("ESP8266Node");
    client.subscribe("factory/actuator/#");
    delay(2000);
  }
}

// ==================== SENSOR READINGS ====================
float readTankLevel() {
  // Read ultrasonic distance (cm)
  digitalWrite(ULTRASONIC_TRIG, LOW);
  delayMicroseconds(2);
  
  digitalWrite(ULTRASONIC_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(ULTRASONIC_TRIG, LOW);
  
  long duration = pulseIn(ULTRASONIC_ECHO, HIGH, 30000);  // Timeout 30ms
  if (duration == 0) return -1;  // Out of range
  
  float distanceCm = duration * 0.034 / 2;
  
  // Convert to tank level percentage (assuming 50cm tank height)
  float tankHeight = 50.0;
  float levelPercent = ((tankHeight - distanceCm) / tankHeight) * 100;
  levelPercent = constrain(levelPercent, 0, 100);
  
  return levelPercent;
}

int readGasSensor() {
  // Average multiple readings for stability
  int total = 0;
  for (int i = 0; i < 5; i++) {
    total += analogRead(MQ2_PIN);
    delay(10);
  }
  return total / 5;
}

float calculateBatteryLevel() {
  // ESP8266 doesn't have battery, simulate based on WiFi signal
  int rssi = WiFi.RSSI();
  // Map RSSI (-100 to -30) to battery (0-100%)
  float battery = map(rssi, -100, -30, 0, 100);
  return constrain(battery, 0, 100);
}

// ==================== SETUP ====================
void setup() {
  Serial.begin(115200);  // Faster baud rate
  
  Serial.println("\n");
  Serial.println("============================================");
  Serial.println("🏭 FOGNET-X Factory IoT Node Initializing...");
  Serial.println("============================================");
  
  // Pin initialization
  pinMode(PIR_ENTRY, INPUT);
  pinMode(PIR_EXIT, INPUT_PULLUP);
  pinMode(ULTRASONIC_TRIG, OUTPUT);
  pinMode(ULTRASONIC_ECHO, INPUT);
  pinMode(RELAY_FAN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  
  // Default states (relays active LOW)
  digitalWrite(RELAY_FAN, HIGH);  // Fan OFF
  digitalWrite(BUZZER_PIN, LOW);  // Buzzer OFF
  
  // Servo initialization
  ventServo.attach(SERVO_VENT);
  ventServo.write(0);  // Start closed
  
  // Sensor initialization
  dht.begin();
  
  Serial.println("✅ Sensors initialized");
  
  // WiFi connection
  setup_wifi();
  
  // MQTT configuration
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);
  
  Serial.println("✅ System Ready!");
  Serial.println("============================================");
}

// ==================== MAIN LOOP ====================
void loop() {
  // Maintain MQTT connection
  if (!client.connected()) {
    reconnect();
  }
  client.loop();
  
  unsigned long currentTime = millis();
  
  // Send sensor data every 2 seconds
  if (currentTime - lastPublishTime >= publishInterval) {
    lastPublishTime = currentTime;
    publishSensorData();
  }
  
  // Heartbeat every 5 seconds
  if (currentTime % heartbeatInterval < 100) {
    client.publish("factory/status/heartbeat", "alive");
  }
  
  delay(100);  // Small delay to prevent watchdog reset
}

// ==================== DATA PUBLISHING ====================
void publishSensorData() {
  // Read all sensors
  float temp = dht.readTemperature();
  float humidity = dht.readHumidity();
  int gasRaw = readGasSensor();
  int pirEntry = digitalRead(PIR_ENTRY);
  float tankLevel = readTankLevel();
  int lightLevel = analogRead(LIGHT_SENSOR);
  
  // Check for alert conditions
  checkAlerts(gasRaw, tankLevel, pirEntry);
  
  // Build JSON payload with ALL enhanced sensor fields
  StaticJsonDocument<512> doc;
  
  // Core device info
  doc["device_id"] = DEVICE_ID;
  doc["device_type"] = DEVICE_TYPE;
  doc["location"] = DEVICE_LOCATION;
  doc["firmware_version"] = "1.0.0";
  doc["capabilities"] = "[\"temperature\", \"humidity\", \"gas\", \"motion\", \"tank_level\", \"light_lux\"]";
  
  // Environmental sensors
  if (!isnan(temp)) doc["temperature"] = temp;
  if (!isnan(humidity)) doc["humidity"] = humidity;
  
  // Gas sensor (convert raw to approximate PPM)
  // MQ2: 0-1023 maps roughly to 0-1000 PPM
  float gasPPM = (gasRaw / 1023.0) * 1000;
  doc["gas"] = round(gasPPM);
  doc["gas_raw"] = gasRaw;
  
  // Tank monitoring
  if (tankLevel >= 0) {
    doc["tank_level"] = round(tankLevel * 10) / 10.0;
  }
  
  // Security sensors
  doc["motion"] = pirEntry ? 1 : 0;
  doc["pir_entry"] = pirEntry ? 1 : 0;
  
  // Light sensor
  float lux = (lightLevel / 1023.0) * 10000;  // Approximate lux
  doc["light_lux"] = round(lux);
  
  // Device health metrics
  doc["battery"] = round(calculateBatteryLevel());
  doc["rssi"] = WiFi.RSSI();
  doc["uptime"] = millis() / 1000;
  
  // Simulated device metrics (for demo)
  doc["device_cpu"] = random(30, 60);
  doc["device_memory"] = random(40, 70);
  doc["network_latency"] = random(30, 80);
  doc["packet_loss"] = random(0, 20) / 10.0;
  
  // Alert status flags
  doc["gas_alert"] = gas_alertActive ? 1 : 0;
  doc["overflow_alert"] = overflowAlertActive ? 1 : 0;
  doc["security_alert"] = securityAlertActive ? 1 : 0;
  
  // Serialize and publish
  String payload;
  serializeJson(doc, payload);
  
  bool published = client.publish("fognetx/sensors", payload.c_str());
  
  if (published) {
    Serial.print("📤 Published: ");
    Serial.print("Temp: " + String(temp) + "°C");
    Serial.print(" | Gas: " + String(round(gasPPM)) + "ppm");
    Serial.print(" | Tank: " + String(round(tankLevel)) + "%");
    Serial.print(" | Motion: " + String(pirEntry ? "DETECTED" : "CLEAR"));
    if (gas_alertActive) Serial.print(" ⚠️ GAS ALERT");
    if (overflowAlertActive) Serial.print(" ⚠️ OVERFLOW ALERT");
    if (securityAlertActive) Serial.print(" 🔒 SECURITY ALERT");
    Serial.println();
  } else {
    Serial.println("❌ Publish failed");
  }
}

// ==================== ALERT LOGIC ====================
void checkAlerts(int gasValue, float tankLevel, int motionDetected) {
  unsigned long now = millis();
  
  // Cooldown check (prevent spam)
  if (now < alertCooldown) return;
  
  // === SECTION A: Gas Storage Tank ===
  if (gasValue > GAS_CRITICAL_MAX) {
    if (!gas_alertActive) {
      Serial.println("🚨 CRITICAL: Gas leak detected!");
      gas_alertActive = true;
      triggerEmergencyResponse();
      alertCooldown = now + 10000;  // 10 second cooldown
    }
  } else if (gasValue < GAS_NORMAL_MAX) {
    gas_alertActive = false;
  }
  
  // Tank overflow detection
  if (tankLevel > 90) {  // 90% full
    if (!overflowAlertActive) {
      Serial.println("🚨 CRITICAL: Tank overflow risk!");
      overflowAlertActive = true;
      alertCooldown = now + 5000;
    }
  } else if (tankLevel < 80) {
    overflowAlertActive = false;
  }
  
  // === SECTION C: Entry Security ===
  if (motionDetected && gas_alertActive) {
    // High priority: human detected during gas leak
    if (!securityAlertActive) {
      Serial.println("🚨🚨 CRITICAL: Human in hazardous area!");
      securityAlertActive = true;
      triggerSecurityAlert();
      alertCooldown = now + 15000;
    }
  } else if (!motionDetected && !gas_alertActive) {
    securityAlertActive = false;
  }
}

// ==================== ACTUATOR CONTROL ====================
void triggerEmergencyResponse() {
  // Automatic response to gas leak
  Serial.println("⚡ Activating emergency response...");
  
  // Turn on exhaust fan
  digitalWrite(RELAY_FAN, LOW);
  Serial.println("  → FAN ON");
  
  // Open vent
  ventServo.write(90);
  Serial.println("  → VENT OPEN");
  
  // Sound local alarm
  tone(BUZZER_PIN, 1000, 500);  // Beep
  
  // Note: Cloud/Fog will also receive the alert and can take additional actions
}

void triggerSecurityAlert() {
  // Rapid beeping for security alert
  for (int i = 0; i < 3; i++) {
    tone(BUZZER_PIN, 1500, 200);
    delay(300);
  }
}

// ==================== WIFI & MQTT ====================
void reconnect() {
  while (!client.connected()) {
    Serial.print("Attempting MQTT connection...");
    
    String clientId = "ESP8266-" + String(DEVICE_ID);
    
    if (client.connect(clientId.c_str())) {
      Serial.println("✅ MQTT Connected!");
      
      // Subscribe to actuator topics
      client.subscribe("factory/actuator/#");
      client.subscribe("fognetx/actuator/#");
      
      Serial.println("📡 Subscribed to actuator topics");
    } else {
      Serial.print("❌ Failed, rc=");
      Serial.print(client.state());
      Serial.println(" Retrying in 2s...");
      delay(2000);
    }
  }
}
